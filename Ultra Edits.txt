const formatTime = (sec) => {
  const s = Math.max(0, Math.floor(sec));
  const m = Math.floor(s / 60);
  const r = s % 60;
  return `${String(m).padStart(2, '0')}:${String(r).padStart(2, '0')}`;
};

          <div className="space-y-3">
            <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-2 gap-1.5">
              {EFFECTS_LIST.map((ef) => (
                <button
                  key={ef.id}
                  onClick={() => {
                    setActiveEffect(ef.id);
                    showToast(`Effect: ${ef.name}`);
                  }}
                  className={`px-2.5 py-2 rounded-xl text-xs font-semibold border text-left transition ${
                    activeEffect === ef.id
                      ? 'border-[#FF7A18] bg-[#FF7A18]/20 text-[#FFB347]'
                      : `${panelBg} hover:border-white/25`
                  }`}
                >
                  {ef.name}
                </button>
              ))}
            </div>
            <div>
              <div className="flex justify-between text-xs mb-1">
                <span>Intensity</span>
                <span className="text-[#FFB347] font-mono">{effectIntensity}%</span>
              </div>
              <input
                type="range"
                min="0"
                max="100"
                value={effectIntensity}
                onChange={(e) => setEffectIntensity(Number(e.target.value))}
                className="w-full accent-[#FF7A18]"
              />
            </div>
          </div>
        )}

        {/* TRANSITIONS */}
        {activeEditorToolTab === 'transitions' && (
          <div className="space-y-2.5">
            <p className={`text-xs ${mutedText}`}>Select transition between V1 clips:</p>
            <div className="grid grid-cols-2 gap-1.5">
              {TRANSITIONS_LIST.map((tr) => (
                <button
                  key={tr}
                  onClick={() => {
                    setActiveTransition(tr);
                    showToast(`Transition set: ${tr}`);
                  }}
                  className={`px-3 py-2 rounded-xl text-xs font-semibold border transition ${
                    activeTransition === tr
                      ? 'border-[#FF7A18] bg-[#FF7A18]/20 text-[#FFB347]'
                      : `${panelBg}`
                  }`}
                >
                  {tr}
                </button>
              ))}
            </div>
          </div>
        )}

        {/* TEXT */}
        {activeEditorToolTab === 'text' && (
          <div className="space-y-3">
            <div>
              <label className="text-xs font-bold block mb-1">Live Caption Input</label>
              <input
                type="text"
                value={captionText}
                onChange={(e) => setCaptionText(e.target.value)}
                placeholder="Type caption or বাংলা টেক্সট..."
                className={`w-full px-3 py-2 rounded-xl border text-xs outline-none focus:border-[#FF7A18] ${panelBg}`}
              />
            </div>
            <div className="flex flex-wrap gap-1.5">
              {[
                { key: 'animated', label: 'Animated Text' },
                { key: 'banglaFont', label: 'Bangla Font (বাংলা)' },
                { key: 'gradient', label: 'Gradient' },
                { key: 'neon', label: 'Neon' },
                { key: 'shadow', label: 'Shadow' },
              ].map((opt) => (
                <button
                  key={opt.key}
                  onClick={() => setTextStyle((s) => ({ ...s, [opt.key]: !s[opt.key] }))}
                  className={`px-2.5 py-1.5 rounded-xl text-[11px] font-bold border ${
                    textStyle[opt.key] ? 'bg-[#FF7A18] text-black border-[#FF7A18]' : panelBg
                  }`}
                >
                  {opt.label}
                </button>
              ))}
            </div>
            <div className="flex items-center justify-between gap-2">
              <label className={`cursor-pointer px-3 py-1.5 rounded-xl border text-xs font-semibold flex items-center gap-1.5 ${panelBg} hover:border-[#FF7A18]`}>
                <Upload className="w-3.5 h-3.5 text-[#FF7A18]" />
                <span>Custom Font Upload</span>
                <input
                  type="file"
                  accept=".ttf,.otf,.woff,.woff2"
                  className="hidden"
                  onChange={(e) => {
                    const f = e.target.files?.[0];
                    if (f) {
                      setTextStyle((s) => ({ ...s, customFontName: f.name }));
                      showToast(`Loaded font: ${f.name}`);
                    }
                  }}
                />
              </label>
              <button
                onClick={() => {
                  setCaptionText('সীমাহীন এডিটিং ✨');
                  setTextStyle((s) => ({ ...s, banglaFont: true }));
                  showToast('Bangla caption applied');
                }}
                className="px-2.5 py-1.5 rounded-xl bg-[#FF7A18]/20 text-[#FFB347] text-[11px] font-bold"
              >
                Sample বাংলা
              </button>
            </div>
          </div>
        )}

        {/* AUDIO */}
        {activeEditorToolTab === 'audio' && (
          <div className="space-y-3">
            <div className="space-y-1.5">
              {MUSIC_TRACKS.map((m) => (
                <button
                  key={m.id}
                  onClick={() => {
                    setAudioTrack((a) => ({ ...a, name: m.title }));
                    showToast(`Music: ${m.title}`);
                  }}
                  className={`w-full px-3 py-2 rounded-xl border flex items-center justify-between text-xs ${
                    audioTrack.name === m.title ? 'border-[#FF7A18] bg-[#FF7A18]/15' : panelBg
                  }`}
                >
                  <span className="font-semibold flex items-center gap-1.5">
                    <Music className="w-3.5 h-3.5 text-[#FF7A18]" />
                    {m.title}
                  </span>
                  <span className="text-[10px] font-mono text-[#FFB347]">{m.bpm} BPM</span>
                </button>
              ))}
            </div>
            <div>
              <div className="flex justify-between text-xs mb-1">
                <span>Volume</span>
                <span className="font-mono text-[#FFB347]">{audioTrack.muted ? 0 : audioTrack.volume}%</span>
              </div>
              <input
                type="range"
                min="0"
                max="100"
                value={audioTrack.muted ? 0 : audioTrack.volume}
                onChange={(e) => setAudioTrack((a) => ({ ...a, volume: Number(e.target.value), muted: false }))}
                className="w-full accent-[#FF7A18]"
              />
            </div>
            <div className="flex flex-wrap gap-1.5">
              <button
                onClick={() => showToast('Audio extracted to A1 track')}
                className={`px-2.5 py-1.5 rounded-xl text-[11px] font-bold border ${panelBg} hover:border-[#FF7A18]`}
              >
                Extract Audio
              </button>
              <button
                onClick={() => {
                  setAudioTrack((a) => ({ ...a, recording: !a.recording }));
                  showToast(audioTrack.recording ? 'Voice recording saved' : 'Voice recording started...');
                }}
                className={`px-2.5 py-1.5 rounded-xl text-[11px] font-bold border flex items-center gap-1 ${
                  audioTrack.recording ? 'bg-red-500 text-white border-red-500' : panelBg
                }`}
              >
                <Mic className="w-3 h-3" />
                {audioTrack.recording ? 'Stop Rec' : 'Voice Recording'}
              </button>
              <button
                onClick={() => setAudioTrack((a) => ({ ...a, noiseReduction: !a.noiseReduction }))}
                className={`px-2.5 py-1.5 rounded-xl text-[11px] font-bold border ${
                  audioTrack.noiseReduction ? 'bg-[#FF7A18] text-black border-[#FF7A18]' : panelBg
                }`}
              >
                Noise Reduction
              </button>
              <button
                onClick={() => setAudioTrack((a) => ({ ...a, fadeIn: !a.fadeIn, fadeOut: !a.fadeOut }))}
                className={`px-2.5 py-1.5 rounded-xl text-[11px] font-bold border ${
                  audioTrack.fadeIn ? 'bg-[#FF7A18]/20 text-[#FFB347] border-[#FF7A18]' : panelBg
                }`}
              >
                Fade In/Out
              </button>
              <button
                onClick={() => showToast('Beat Sync aligned clips to BPM')}
                className={`px-2.5 py-1.5 rounded-xl text-[11px] font-bold border ${panelBg} hover:border-[#FF7A18]`}
              >
                Beat Sync
              </button>
              <button
                onClick={() => {
                  setAudioTrack((a) => ({ ...a, duration: Math.max(5, a.duration - 2) }));
                  showToast('Audio trimmed -2s');
                }}
                className={`px-2.5 py-1.5 rounded-xl text-[11px] font-bold border ${panelBg} hover:border-[#FF7A18]`}
              >
                Audio Trimming
              </button>
            </div>
          </div>
        )}

        {/* STICKERS */}
        {activeEditorToolTab === 'stickers' && (
          <div className="grid grid-cols-2 gap-2">
            {STICKER_TYPES.map((st) => (
              <button
                key={st.id}
                onClick={() => handleAddSticker(st)}
                className={`p-2.5 rounded-xl border ${panelBg} hover:border-[#FF7A18] flex items-center gap-2 text-xs font-semibold transition`}
              >
                <span className="text-base">{st.icon}</span>
                <span className="truncate">{st.label}</span>
              </button>
            ))}
          </div>
        )}

        {/* SPEED */}
        {activeEditorToolTab === 'speed' && (
          <div className="space-y-3">
            <div className="flex gap-2">
              {[0.5, 1, 2].map((sp) => (
                <button
                  key={sp}
                  onClick={() => {
                    setSpeed(sp);
                    showToast(`Playback speed: ${sp}x`);
                  }}
                  className={`flex-1 py-2 rounded-xl text-xs font-extrabold border transition ${
                    speed === sp ? 'bg-[#FF7A18] text-black border-[#FF7A18]' : panelBg
                  }`}
                >
                  {sp}x {sp === 0.5 ? '(Slow-Mo)' : ''}
                </button>
              ))}
            </div>
            <div>
              <div className="flex justify-between text-xs mb-1">
                <span>Speed Ramp</span>
                <span className="font-mono text-[#FFB347]">{speedRamp.toFixed(1)}x</span>
              </div>
              <input
                type="range"
                min="0.2"
                max="3.0"
                step="0.1"
                value={speedRamp}
                onChange={(e) => setSpeedRamp(Number(e.target.value))}
                className="w-full accent-[#FF7A18]"
              />
            </div>
          </div>
        )}

        {/* IMPORT (Mobile shortcut) */}
        {activeEditorToolTab === 'import' && (
          <div className="space-y-2.5">
            <p className={`text-xs ${mutedText}`}>Multi-Video Import (mp4, mov, mkv, avi)</p>
            <label className="w-full py-3 rounded-xl bg-[#FF7A18] text-black font-bold text-xs flex items-center justify-center gap-2 cursor-pointer">
              <Upload className="w-4 h-4" />
              <span>Select Video Files</span>
              <input
                type="file"
                multiple
                accept=".mp4,.mov,.mkv,.avi,video/*"
                className="hidden"
                onChange={(e) => handleImportVideos(e.target.files)}
              />
            </label>
          </div>
        )}
      </div>
    );

    return (
      <div className="space-y-3 pb-4">
        {/* 3-Panel Layout on Desktop (1024px+), Stacked on Mobile */}
        <div className="grid grid-cols-1 lg:grid-cols-12 gap-4 items-start">
          {/* Left Media Panel (Desktop) */}
          <div className="hidden lg:block lg:col-span-3">{leftMediaPanel}</div>

          {/* Center Preview + Tools Row + Timeline */}
          <div className="lg:col-span-6 space-y-3">
            {renderVideoPreview()}

            {/* Quick Tools Row */}
            <div className={`flex items-center justify-between gap-1.5 px-2.5 py-2 rounded-2xl border overflow-x-auto ${cardBg}`}>
              {[
                { icon: audioTrack.muted ? VolumeX : Volume2, label: 'Volume', action: () => setActiveEditorToolTab('audio') },
                { icon: VolumeX, label: audioTrack.muted ? 'Unmute' : 'Mute', action: () => setAudioTrack((a) => ({ ...a, muted: !a.muted })) },
                {
                  icon: Crop,
                  label: `Crop (${aspectCrop})`,
                  action: () => {
                    const order = ['16:9', '9:16', '1:1'];
                    const next = order[(order.indexOf(aspectCrop) + 1) % order.length];
                    setAspectCrop(next);
                    showToast(`Crop: ${next}`);
                  },
                },
                { icon: Maximize2, label: 'Fullscreen', action: () => setIsFullscreen(true) },
                { icon: Scissors, label: 'Cut', action: handleSplit },
                {
                  icon: RotateCw,
                  label: 'Rotate',
                  action: () => {
                    setRotation((r) => (r + 90) % 360);
                    showToast('Rotated 90°');
                  },
                },
              ].map((btn) => (
                <button
                  key={btn.label}
                  onClick={btn.action}
                  className="flex items-center gap-1 px-2.5 py-1.5 rounded-xl text-[11px] font-semibold hover:bg-[#FF7A18]/20 hover:text-[#FFB347] whitespace-nowrap transition"
                >
                  <btn.icon className="w-3.5 h-3.5 text-[#FF7A18]" />
                  <span>{btn.label}</span>
                </button>
              ))}
            </div>

            {/* Timeline Controls Bar */}
            <div className={`p-2.5 rounded-2xl border space-y-2.5 ${cardBg}`}>
              <div className="flex flex-wrap items-center justify-between gap-2">
                <div className="flex flex-wrap items-center gap-1">
                  <button onClick={handleSplit} className="px-2.5 py-1.5 rounded-lg bg-[#FF7A18] text-black text-[11px] font-extrabold flex items-center gap-1">
                    <Scissors className="w-3 h-3" /> Split
                  </button>
                  <button onClick={handleTrim} className={`px-2.5 py-1.5 rounded-lg border text-[11px] font-bold ${panelBg} hover:border-[#FF7A18]`}>
                    Trim
                  </button>
                  <button onClick={handleDuplicate} className={`px-2 py-1.5 rounded-lg border text-[11px] font-bold ${panelBg} hover:border-[#FF7A18]`} title="Duplicate">
                    <Copy className="w-3.5 h-3.5" />
                  </button>
                  <button onClick={() => setFlipped((f) => !f)} className={`px-2 py-1.5 rounded-lg border text-[11px] font-bold ${panelBg} hover:border-[#FF7A18]`} title="Flip">
                    <FlipHorizontal className="w-3.5 h-3.5" />
                  </button>
                  <button onClick={handleUndo} className={`px-2 py-1.5 rounded-lg border text-[11px] font-bold ${panelBg} hover:border-[#FF7A18]`} title="Undo">
                    <Undo2 className="w-3.5 h-3.5" />
                  </button>
                  <button onClick={handleRedo} className={`px-2 py-1.5 rounded-lg border text-[11px] font-bold ${panelBg} hover:border-[#FF7A18]`} title="Redo">
                    <Redo2 className="w-3.5 h-3.5" />
                  </button>
                </div>

                <div className="flex items-center gap-2">
                  <button
                    onClick={() => setSnapEnabled((s) => !s)}
                    className={`px-2 py-1 rounded-lg text-[10px] font-bold flex items-center gap-1 border ${
                      snapEnabled ? 'border-[#FF7A18] text-[#FFB347] bg-[#FF7A18]/15' : panelBg
                    }`}
                  >
                    <Magnet className="w-3 h-3" /> Snap {snapEnabled ? 'ON' : 'OFF'}
                  </button>
                  <div className="flex items-center gap-1.5">
                    <span className="text-[10px] text-zinc-400">Zoom</span>
                    <input
                      type="range"
                      min="10"
                      max="40"
                      value={timelineZoom}
                      onChange={(e) => setTimelineZoom(Number(e.target.value))}
                      className="w-20 accent-[#FF7A18]"
                    />
                  </div>
                </div>
              </div>

              {/* Scrollable Multi-Track Timeline */}
              <div className="overflow-x-auto pb-2 relative">
                <div
                  className="relative select-none"
                  style={{ width: `${Math.max(30, totalDuration) * timelineZoom + 80}px` }}
                >
                  {/* Ruler: 0s, 5s, 10s, 15s, 20s, 25s, 30s */}
                  <div
                    onClick={(e) => {
                      const rect = e.currentTarget.getBoundingClientRect();
                      const sec = Math.max(0, Math.min(totalDuration, (e.clientX - rect.left - 44) / timelineZoom));
                      setCurrentTime(snapEnabled ? Math.round(sec) : sec);
                    }}
                    className="h-6 flex items-end border-b border-white/10 pl-11 cursor-pointer relative"
                  >
                    {rulerMarks.map((sec) => (
                      <div
                        key={sec}
                        className="absolute bottom-0 text-[10px] font-mono text-zinc-400 border-l border-zinc-600 pl-1"
                        style={{ left: `${44 + sec * timelineZoom}px` }}
                      >
                        {sec}s
                      </div>
                    ))}
                  </div>

                  {/* Playhead Line */}
                  <div
                    className="absolute top-0 bottom-0 w-0.5 bg-[#FFB347] z-20 pointer-events-none shadow-[0_0_8px_#FF7A18]"
                    style={{ left: `${44 + currentTime * timelineZoom}px` }}
                  >
                    <div className="w-2.5 h-2.5 -ml-[4px] rounded-full bg-[#FF7A18]" />
                  </div>

                  {/* Track 1: V1 Video */}
                  <div className="flex items-center h-11 border-b border-white/5 relative mt-1">
                    <span className="w-11 shrink-0 text-[10px] font-extrabold text-[#FFB347]">V1</span>
                    <div className="relative flex-1 h-9 flex items-center">
                      {clips.map((c, i) => (
                        <React.Fragment key={c.id}>
                          <div
                            onClick={() => setSelectedClipId(c.id)}
                            style={{ width: `${c.duration * timelineZoom}px` }}
                            className={`h-8 rounded-lg bg-gradient-to-r ${c.color} px-2 flex items-center justify-between cursor-pointer border-2 transition-all duration-150 shrink-0 ${
                              selectedClipId === c.id ? 'border-white shadow-md' : 'border-transparent opacity-90'
                            }`}
                          >
                            <span className="text-[10px] font-bold text-black truncate">{c.name}</span>
                            <span className="text-[9px] font-mono font-bold text-black/80 ml-1">{formatTime(c.duration)}</span>
                          </div>
                          {i < clips.length - 1 && (
                            <div
                              onClick={() => setActiveEditorToolTab('transitions')}
                              className="w-4 h-6 -mx-2 z-10 rounded bg-black border border-[#FF7A18] flex items-center justify-center text-[8px] font-bold text-[#FFB347] cursor-pointer shrink-0"
                              title={`Transition: ${activeTransition}`}
                            >
                              ⚡
                            </div>
                          )}
                        </React.Fragment>
                      ))}
                      {/* Keyframe Dots */}
                      {keyframes.map((kf) => (
                        <div
                          key={kf}
                          className="absolute bottom-0.5 w-2 h-2 rotate-45 bg-white border border-[#FF7A18] z-10 pointer-events-none"
                          style={{ left: `${kf * timelineZoom}px` }}
                          title={`Keyframe at ${kf}s`}
                        />
                      ))}
                    </div>
                  </div>

                  {/* Track 2: A1 Audio Waveform */}
                  <div className="flex items-center h-10 border-b border-white/5 relative">
                    <span className="w-11 shrink-0 text-[10px] font-extrabold text-emerald-400">A1</span>
                    <div
                      style={{ width: `${audioTrack.duration * timelineZoom}px` }}
                      className="h-7 rounded-lg bg-emerald-500/20 border border-emerald-500/40 px-2 flex items-center gap-0.5 overflow-hidden"
                    >
                      {Array.from({ length: 36 }).map((_, idx) => (
                        <div
                          key={idx}
                          className="w-1 bg-emerald-400/80 rounded-full shrink-0"
                          style={{ height: `${25 + ((idx * 17) % 65)}%` }}
                        />
                      ))}
                      <span className="text-[9px] text-emerald-200 font-semibold ml-2 truncate">{audioTrack.name}</span>
                    </div>
                  </div>

                  {/* Track 3: T Text */}
                  <div className="flex items-center h-9 border-b border-white/5 relative">
                    <span className="w-11 shrink-0 text-[10px] font-extrabold text-sky-400">T</span>
                    {textTrackItems.map((t) => (
                      <div
                        key={t.id}
                        style={{ marginLeft: `${t.start * timelineZoom}px`, width: `${t.duration * timelineZoom}px` }}
                        className="h-6 rounded-md bg-sky-500/25 border border-sky-400/50 px-2 flex items-center text-[10px] text-sky-200 truncate"
                      >
                        {captionText || t.label}
                      </div>
                    ))}
                  </div>

                  {/* Track 4: S Sticker */}
                  <div className="flex items-center h-9 relative">
                    <span className="w-11 shrink-0 text-[10px] font-extrabold text-purple-400">S</span>
                    <div className="relative flex-1 h-7 flex items-center">
                      {stickerItems.map((s) => (
                        <div
                          key={s.id}
                          style={{ left: `${s.start * timelineZoom}px`, width: `${s.duration * timelineZoom}px` }}
                          className="absolute h-6 rounded-md bg-purple-500/25 border border-purple-400/50 px-1.5 flex items-center gap-1 text-[10px] text-purple-200 truncate"
                        >
                          <span>{s.icon}</span>
                          <span className="truncate">{s.label}</span>
                        </div>
                      ))}
                    </div>
                  </div>
                </div>
              </div>
            </div>
          </div>

          {/* Right Tools Panel */}
          <div className="lg:col-span-3 space-y-3">
            {rightToolsPanel}
            <div className="lg:hidden">{leftMediaPanel}</div>
          </div>
        </div>
      </div>
    );
  };

  // ==================== SCREEN 4: PRO TOOLS ====================
  const renderProTools = () => (
    <div className="space-y-5 pb-6">
      <div className="grid grid-cols-1 lg:grid-cols-12 gap-4 items-start">
        <div className="lg:col-span-5">{renderVideoPreview(true)}</div>

        <div className="lg:col-span-7 grid grid-cols-1 sm:grid-cols-2 gap-3.5">
          {/* Magic Remover */}
          <div className={`p-4 rounded-2xl border space-y-2.5 ${cardBg}`}>
            <div className="flex items-center justify-between">
              <h3 className="text-xs font-extrabold uppercase text-[#FFB347] flex items-center gap-1.5">
                <Eraser className="w-4 h-4 text-[#FF7A18]" /> Magic Object Remover
              </h3>
              {objectRemoved && <span className="text-[10px] text-emerald-400 font-bold">Object Removed</span>}
            </div>
            <div>
              <div className="flex justify-between text-xs mb-1">
                <span>Brush Size</span>
                <span className="font-mono text-[#FFB347]">{brushSize}px</span>
              </div>
              <input type="range" min="5" max="80" value={brushSize} onChange={(e) => setBrushSize(Number(e.target.value))} className="w-full accent-[#FF7A18]" />
            </div>
            <button
              onClick={() => {
                setObjectRemoved(true);
                showToast('Object Removed cleanly');
              }}
              className="w-full py-2 rounded-xl bg-[#FF7A18] text-black font-bold text-xs"
            >
              Erase Object
            </button>
          </div>

          {/* Chroma Key */}
          <div className={`p-4 rounded-2xl border space-y-2.5 ${cardBg}`}>
            <h3 className="text-xs font-extrabold uppercase text-[#FFB347] flex items-center gap-1.5">
              <Palette className="w-4 h-4 text-[#FF7A18]" /> Chroma Key / Green Screen
            </h3>
            <div className="flex gap-1.5">
              {['None', 'Green', 'Blue', 'Red'].map((col) => (
                <button
                  key={col}
                  onClick={() => {
                    setChromaColor(col);
                    showToast(`Chroma Key: ${col}`);
                  }}
                  className={`flex-1 py-1.5 rounded-xl text-[11px] font-bold border ${
                    chromaColor === col ? 'border-[#FF7A18] bg-[#FF7A18] text-black' : panelBg
                  }`}
                >
                  {col}
                </button>
              ))}
            </div>
            <div className="grid grid-cols-2 gap-2 text-[11px]">
              <div>
                <span>Tolerance ({chromaTolerance}%)</span>
                <input type="range" min="0" max="100" value={chromaTolerance} onChange={(e) => setChromaTolerance(Number(e.target.value))} className="w-full accent-[#FF7A18]" />
              </div>
              <div>
                <span>Feather ({chromaFeather}%)</span>

                className={`px-3.5 py-1.5 rounded-xl text-xs font-bold flex items-center gap-1.5 transition ${
                  activeTab === item.id
                    ? 'bg-gradient-to-r from-[#FF7A18] to-[#FFB347] text-black shadow-sm'
                    : 'hover:text-[#FF7A18]'
                }`}
              >
                <item.icon className="w-3.5 h-3.5" />
                <span>{item.label}</span>
              </button>
            ))}
          </nav>

          <button
            onClick={() => setActiveTab('premium')}
            className="px-3 py-1.5 rounded-xl bg-[#FF7A18]/15 border border-[#FF7A18]/40 text-[#FFB347] text-xs font-bold flex items-center gap-1.5 shrink-0"
          >
            <Crown className="w-3.5 h-3.5" />
            <span>{isPremium ? 'VIP' : 'Go Pro'}</span>
          </button>
        </div>
      </header>

      {/* Main Content */}
      <main className="flex-1 max-w-7xl w-full mx-auto px-3 sm:px-5 pt-4 pb-24 md:pb-8">
        {activeTab === 'home' && renderHome()}
        {activeTab === 'editor' && renderEditor()}
        {activeTab === 'pro' && renderProTools()}
        {activeTab === 'export' && renderExport()}
        {activeTab === 'premium' && renderPremium()}
        {activeTab === 'settings' && renderSettings()}
      </main>

      {/* Mobile Bottom Navigation (<768px / <600px phone layout) */}
      <nav className={`md:hidden fixed bottom-0 inset-x-0 z-40 border-t backdrop-blur-xl px-1 py-1.5 flex justify-around ${darkMode ? 'bg-[#0B0B0B]/95 border-white/10' : 'bg-white/95 border-black/10'}`}>
        {navItems.map((item) => {
          const active = activeTab === item.id;
          return (
            <button
              key={item.id}
              onClick={() => setActiveTab(item.id)}
              className={`flex flex-col items-center gap-0.5 px-2 py-1 rounded-xl transition ${
                active ? 'text-[#FF7A18]' : mutedText
              }`}
            >
              <item.icon className={`w-4 h-4 ${active ? 'scale-110' : ''} transition-transform`} />
              <span className="text-[10px] font-bold">{item.label}</span>
            </button>
          );
        })}
      </nav>

      {/* Global Toast */}
      {toast && (
        <div className="fixed bottom-20 md:bottom-6 left-1/2 -translate-x-1/2 z-50 px-4 py-2.5 rounded-2xl bg-[#161616] text-white border border-[#FF7A18]/60 shadow-[0_0_25px_rgba(255,122,24,0.35)] text-xs font-bold flex items-center gap-2">
          <Sparkles className="w-3.5 h-3.5 text-[#FFB347]" />
          <span>{toast}</span>
        </div>
      )}
    </div>
  );
}
